Automate via Google Sheets

To sync directly to your specific spreadsheet automatically:

  1. Open your Target Google Sheet.
  2. Go to Extensions > Apps Script.
  3. Paste the code below, overwriting any existing code: 
    function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var studentId = e.parameter.studentId;
  
  var data = sheet.getRange("A:A").getValues();
  for (var i = 0; i < data.length; i++) {
    if (data[i][0] == studentId) {
      return ContentService.createTextOutput(JSON.stringify({"status": "duplicate"})).setMimeType(ContentService.MimeType.JSON);
    }
  }

  var rowData = [
    studentId, e.parameter.weight, e.parameter.height, 
    e.parameter.bmi, e.parameter.bmiCategoryAsian, e.parameter.muac, 
    e.parameter.waist, e.parameter.hip, e.parameter.whr, 
    e.parameter.triceps, e.parameter.biceps, e.parameter.subscapular, 
    e.parameter.suprailiac, new Date()
  ];
  sheet.appendRow(rowData);
  return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON);
}
  4. Click Deploy > New Deployment. Select Web App.
  5. Set "Who has access" to Anyone. Click Deploy.
